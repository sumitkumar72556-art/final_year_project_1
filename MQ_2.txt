#define MQ2_PIN 32
#define MQ7_PIN 35
#define BUZZER_PIN 25
#define LED_PIN 26

void setup() {
  Serial.begin(115200);

  analogReadResolution(12);
  analogSetPinAttenuation(MQ2_PIN, ADC_11db);
  analogSetPinAttenuation(MQ7_PIN, ADC_11db);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);

  digitalWrite(BUZZER_PIN, LOW);
  digitalWrite(LED_PIN, LOW);

  Serial.println("SMART MINE WORKER SAFETY SYSTEM");
  Serial.println("Warming up sensors for 20 seconds...");
  delay(20000);   // MQ sensors need warm-up
}

void loop() {
  int mq2 = analogRead(MQ2_PIN);
  int mq7 = analogRead(MQ7_PIN);

  Serial.println("----------------------------");
  Serial.print("MQ-2: ");
  Serial.println(mq2);

  Serial.print("MQ-7: ");
  Serial.println(mq7);

  if (mq2 >= 1200 || mq7 >= 1000) {
    Serial.println("*** DANGER! GAS DETECTED ***");

    digitalWrite(LED_PIN, HIGH);
    digitalWrite(BUZZER_PIN, HIGH);
  }
  else {
    Serial.println("SAFE");

    digitalWrite(LED_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);
  }

  delay(1000);
}
